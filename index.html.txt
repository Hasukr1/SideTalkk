  <!-- Firebase Integration -->
  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";
    import { getDatabase, ref, push, onChildAdded } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-database.js";

    // PASTE YOUR FIREBASE CONFIG HERE IN STEP 2
    const firebaseConfig = {
      databaseURL: "https://YOUR-DATABASE-NAME-default-rtdb.firebaseio.com"
    };

    const app = initializeApp(firebaseConfig);
    const db = getDatabase(app);
    const messagesRef = ref(db, 'messages');

    let username = localStorage.getItem('uc_user') || 'User_' + Math.floor(Math.random() * 1000);
    let avatarSeed = localStorage.getItem('uc_seed') || username;

    document.getElementById('usernameDisplay').innerText = username;
    document.getElementById('myAvatar').src = `https://api.dicebear.com/7.x/bottts/svg?seed=${avatarSeed}`;

    // Click Avatar to change Name
    document.getElementById('myAvatar').onclick = () => {
      const newName = prompt("Enter your new display name:", username);
      if (newName) {
        username = newName;
        avatarSeed = newName;
        localStorage.setItem('uc_user', username);
        localStorage.setItem('uc_seed', avatarSeed);
        document.getElementById('usernameDisplay').innerText = username;
        document.getElementById('myAvatar').src = `https://api.dicebear.com/7.x/bottts/svg?seed=${avatarSeed}`;
      }
    };

    // Send Message
    document.getElementById('chatForm').addEventListener('submit', (e) => {
      e.preventDefault();
      const input = document.getElementById('msgInput');
      const text = input.value.trim();
      if (!text) return;

      push(messagesRef, {
        user: username,
        seed: avatarSeed,
        text: text,
        time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
      });

      input.value = '';
    });

    // Listen for incoming messages live
    onChildAdded(messagesRef, (snapshot) => {
      const data = snapshot.val();
      const isMe = data.user === username;

      const isImage = data.text.match(/\.(jpeg|jpg|gif|png|webp)$/i) != null || data.text.startsWith('data:image/');

      const msgHTML = `
        <div class="msg-group ${isMe ? 'sent' : 'received'}">
          <img src="https://api.dicebear.com/7.x/bottts/svg?seed=${data.seed}" class="msg-avatar">
          <div class="msg-body">
            <div class="msg-meta">
              <span>${data.user}</span>
              <span>${data.time}</span>
            </div>
            <div class="bubble">
              ${isImage ? `<img src="${data.text}">` : data.text}
            </div>
          </div>
        </div>
      `;

      const chatBox = document.getElementById('chatBox');
      chatBox.insertAdjacentHTML('beforeend', msgHTML);
      chatBox.scrollTop = chatBox.scrollHeight;
    });
  </script>
